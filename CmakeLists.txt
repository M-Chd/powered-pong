cmake_minimum_required(VERSION 3.10)
project(PoweredPong)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

add_executable(PoweredPong
    src/main.cpp

    src/entities/ball.cpp
    src/entities/player.cpp
    src/entities/activeEffect.cpp
    src/entities/item.cpp

    src/core/window.cpp
    src/core/game.cpp
    src/core/board.cpp
    src/core/inputManager.cpp
    src/core/match.cpp
    src/core/aiController.cpp

    src/util/vector2.cpp

    src/ui/debugOverlay.cpp
    src/ui/inputIP.cpp
    src/ui/menu.cpp
    src/ui/menuManager.cpp
    src/ui/scoreBoard.cpp

    src/net/networkClient.cpp
    src/net/networkManager.cpp
    src/net/networkServer.cpp

    )

target_include_directories(PoweredPong PRIVATE
    ${CMAKE_CURRENT_SOURCE_DIR}/src 
    ${CMAKE_CURRENT_SOURCE_DIR}/NetCommon
)

find_package(SDL2 CONFIG REQUIRED)
find_package(SDL2_ttf CONFIG REQUIRED)
find_package(SDL2_image CONFIG REQUIRED)
find_package(asio CONFIG REQUIRED)

target_link_libraries(PoweredPong PRIVATE SDL2::SDL2 SDL2::SDL2main SDL2_ttf::SDL2_ttf SDL2_image::SDL2_image asio::asio)
